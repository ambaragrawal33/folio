async function measure(name, selectors) {
  const result = await page.evaluate(selectors => {
    const elements = {};
    for (const [key, selector] of Object.entries(selectors)) {
      const el = document.querySelector(selector);
      if (!el) continue;
      const r = el.getBoundingClientRect(), s = getComputedStyle(el);
      elements[key] = { x: r.x, y: r.y, width: r.width, height: r.height, fontFamily: s.fontFamily, fontSize: s.fontSize, lineHeight: s.lineHeight, fontWeight: s.fontWeight, color: s.color, background: s.backgroundColor, border: s.border, borderRadius: s.borderRadius, padding: s.padding, gap: s.gap, outline: s.outline, outlineOffset: s.outlineOffset };
    }
    return { viewport: { width: innerWidth, height: innerHeight }, theme: document.documentElement.dataset.theme, scrollWidth: document.documentElement.scrollWidth, scrollHeight: document.documentElement.scrollHeight, independentScrollers: [...document.querySelectorAll('*')].filter(el => { const s = getComputedStyle(el); return /^(auto|scroll)$/.test(s.overflowY) && el.scrollHeight > el.clientHeight + 1; }).map(el => ({ tag: el.tagName, className: el.className })), elements };
  }, selectors);
  const axe = await new AxeBuilder({ page }).withTags(['wcag2a', 'wcag2aa', 'wcag21aa']).analyze();
  result.axeViolations = axe.violations.map(v => ({ id: v.id, nodes: v.nodes.map(n => n.failureSummary) }));
  evidence.captures.push({ name, ...result });
  expect(result.scrollWidth).toBe(result.viewport.width);
  expect(result.axeViolations).toEqual([]);
  expect(result.independentScrollers).toEqual([]);
}
