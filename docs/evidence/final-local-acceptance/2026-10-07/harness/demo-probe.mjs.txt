import {writeFile} from 'node:fs/promises';import {createRequire} from 'node:module';import assert from 'node:assert/strict';const require=createRequire(new URL('../../package.json',import.meta.url));const {chromium,expect}=require('@playwright/test');const AxeBuilder=require('@axe-core/playwright').default;const browser=await chromium.launch();try{const ctx=await browser.newContext({viewport:{width:1440,height:1024}});const page=await ctx.newPage();await page.goto('http://127.0.0.1:5180/auth/login?theme=dark');await page.getByRole('button',{name:'Explore read-only demo',exact:true}).click();await expect(page.getByLabel('Read-only demo')).toBeVisible();const geometry=await page.evaluate(()=>({viewport:innerWidth,height:innerHeight,document:document.documentElement.scrollWidth,independentScroll:[...document.querySelectorAll('body *')].filter(e=>['auto','scroll'].includes(getComputedStyle(e).overflowY)&&e.scrollHeight>e.clientHeight).map(e=>e.className)}));const axe=await new AxeBuilder({page}).withTags(['wcag2a','wcag2aa','wcag21aa']).analyze();assert.equal(geometry.document,1440);assert.equal(geometry.independentScroll.length,0);assert.equal(axe.violations.length,0);await page.screenshot({path:'.local/final-acceptance/visual/demo-readonly-dark-1440-viewport.png'});await page.screenshot({path:'.local/final-acceptance/visual/demo-readonly-dark-1440.png',fullPage:true});await writeFile('.local/final-acceptance/demo-audit.json',JSON.stringify({name:'demo-readonly-dark-1440',geometry,axeViolations:axe.violations},null,2));console.log('Actual1440×1024 demo capture,zero axe/overflow/independent scroll PASS');}finally{await browser.close();}
